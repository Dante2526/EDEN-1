   439→  // Helper to execute client-side GitHub and web tools
   440→  const executeGitHubTool = async (
   441→    name: string,
   442→    args: any,
   443→    currentRepo: ActiveRepoState | null
   444→  ): Promise<{ result: string; editedFile?: MessageEditedFile }> => {
   445→    if (name === 'web_search' || name === 'search_web') {
   446→      try {
   447→        const query = args.query || args.q || '';
   448→        const res = await fetch(`/api/search?q=${encodeURIComponent(query)}`);
   449→        if (!res.ok) throw new Error(`Falha na busca web (${res.status})`);
   450→        const searchResults = await res.json();
   451→        return { result: JSON.stringify(searchResults) };
   452→      } catch (err: any) {
   453→        return { result: JSON.stringify({ error: err?.message || `Erro ao pesquisar ${args.query}` }) };
   454→      }
   455→    }
   456→
   457→    if (name === 'generate_image') {
   458→      try {
   459→        const prompt = args.prompt || '';
   460→        const aspect_ratio = args.aspect_ratio || '1:1';
   461→        const res = await fetch('/api/image', {
   462→          method: 'POST',
   463→          headers: { 'Content-Type': 'application/json' },
   464→          body: JSON.stringify({ prompt, aspect_ratio }),
   465→        });
   466→        if (!res.ok) throw new Error(`Falha ao gerar imagem (${res.status})`);
   467→        const imgData = await res.json();
   468→        return {
   469→          result: JSON.stringify({
   470→            status: 'success',
   471→            url: imgData.url,
   472→            prompt: imgData.prompt,
   473→            markdown: imgData.markdown,
   474→            instruction:
   475→              `A imagem já foi pré-renderizada e está disponível no servidor. Apresente-a usando exatamente: ${imgData.markdown} sem quebrar linha entre [ e (.`,
   476→          }),
   477→        };
   478→      } catch (err: any) {
   479→        return { result: JSON.stringify({ error: err?.message || 'Erro ao gerar imagem' }) };
   480→      }
   481→    }
   482→
   483→    if (!currentRepo) {
   484→      return {
   485→        result: JSON.stringify({
   486→          error: 'Nenhum repositório GitHub está vinculado a esta conversa. Selecione um repositório no menu superior.',
   487→        }),
   488→      };
   489→    }
   490→
   491→    if (name === 'edit_file') {
   492→      const { path, content = '', type = 'modified' } = args;
   493→
   494→      if (!path) {
   495→        return {
   496→          result: JSON.stringify({ error: 'Parâmetro path é obrigatório para edit_file.' }),
   497→        };
   498→      }
   499→
   500→      let originalContent: string | undefined = undefined;
   501→
   502→      // Try fetching original file content if modifying or deleting
   503→      if (type === 'modified' || type === 'deleted') {
   504→        try {
   505→          originalContent = await fetchFileContent(
   506→            null,
   507→            currentRepo.owner,
   508→            currentRepo.repo,
   509→            path,
   510→            currentRepo.branch
   511→          );
   512→        } catch {
   513→          // New file or unable to read original
   514→        }
   515→      }
   516→
   517→      await addChange({
   518→        path,
   519→        repo: currentRepo.fullName,
   520→        branch: currentRepo.branch,
   521→        type: (type as ChangeType) || 'modified',
   522→        originalContent,
   523→        newContent: content || '',
   524→      });
   525→
   526→      // Show toast
   527→      setToastNotification({
   528→        message: `IA editou ${path} — revise em Source Control`,
   529→        actionLabel: 'Ver Diff',
   530→        onAction: () => {
   531→          handleViewDiffForPath(path);
   532→        },
   533→      });
   534→
   535→      setTimeout(() => {
   536→        setToastNotification(null);
   537→      }, 5000);
   538→
   539→      const editedFile: MessageEditedFile = {
   540→        path,
   541→        type: (type as ChangeType) || 'modified',
   542→      };
   543→
   544→      return {
   545→        result: JSON.stringify({
   546→          success: true,
   547→          message: `Arquivo '${path}' alterado com sucesso e salvo no Source Control local para revisão do usuário.`,
   548→        }),
   549→        editedFile,
   550→      };
   551→    }
   552→
   553→    if (name === 'read_file') {
   554→      try {
   555→        const fileContent = await fetchFileContent(
   556→          null,
   557→          currentRepo.owner,
   558→          currentRepo.repo,
   559→          args.path,
   560→          args.branch || currentRepo.branch
   561→        );
   562→        return { result: fileContent };
   563→      } catch (err: any) {
   564→        return { result: JSON.stringify({ error: err?.message || `Erro ao ler arquivo ${args.path}` }) };
   565→      }
   566→    }
   567→
   568→    if (name === 'list_files') {
   569→      try {
   570→        const contents = await fetchRepoContents(
   571→          null,
   572→          currentRepo.owner,
   573→          currentRepo.repo,
   574→          args.path || '',
   575→          args.branch || currentRepo.branch
   576→        );
   577→        return { result: JSON.stringify(contents.map((c) => ({ name: c.name, path: c.path, type: c.type }))) };
   578→      } catch (err: any) {
   579→        return { result: JSON.stringify({ error: err?.message || `Erro ao listar diretório ${args.path}` }) };
   580→      }
   581→    }
   582→
   583→    if (name === 'search_code') {
   584→      try {
   585→        const items = await searchCode(
   586→          null,
   587→          args.query,
   588→          currentRepo.owner,
   589→          currentRepo.repo
   590→        );
   591→        return {
   592→          result: JSON.stringify(items),
   593→        };
   594→      } catch (err: any) {
