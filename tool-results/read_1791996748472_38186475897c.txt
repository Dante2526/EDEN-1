     7→     7→function maxTokensFor(effort: string): number {
     8→     8→  if (effort === 'max') return 12288;
     9→     9→  if (effort === 'high' || effort === 'medium') return 8192;
    10→    10→  return 4096;
    11→    11→}

    13→    13→// ===========================================================================
    14→    14→// Transformador SSE: extrai <think>...</think> do delta.content da Cloudflare
    15→    15→// e reescreve como delta.reasoning_content, mantendo o resto em delta.content.
    16→    16→//
    17→    17→// Por que isso existe?
    18→    18→//   - O DeepSeek R1-Distill-Qwen-32B da Cloudflare retorna o raciocínio embutido
    19→    19→//     em `delta.content` entre tags <think>...</think>.
    20→    20→//   - O frontend do Synap já sabe separar `delta.reasoning_content` (mostrado na
    21→    21→//     Safe Thinking Orb e ignorado pelo TTS) de `delta.content` (resposta final).
    22→    22→//   - Então interceptamos os chunks SSE aqui no backend e "traduzimos" o formato
    23→    23→//     da Cloudflare para o formato NVIDIA/GLM, deixando o frontend intacto.
    24→    24→//
    25→    25→// Como funciona:
    26→    26→//   - Mantemos um buffer acumulando o `content` de todos os chunks recebidos.
    27→    27→//   - A cada novo chunk, processamos o buffer acumulado procurando por <think> e
    28→    28→//     </think>, emitindo deltas de reasoning ou de content conforme apropriado.
    29→    29→//   - Sempre preservamos o texto incompleto (ex.: "<thi" sem fechar) para o
    30→    30→//     próximo chunk.
    31→    31→//   - No fim do stream (DONE), fazemos um flush final.
    32→    32→// ===========================================================================
    33→    33→class CloudflareThinkSplitter {
    34→    34→  private contentBuffer: string = '';
    35→    35→  private inThinking: boolean = false;
    36→    36→  private thinkTagEmitted: boolean = false;
    37→    37→
    38→    38→  /** Processa um chunk de content (string) e retorna os SSEs a emitir. */
    39→    39→  feed(contentDelta: string): { reasoningDelta?: string; contentDelta?: string } {
    40→    40→    if (!contentDelta) return {};
    41→    41→    this.contentBuffer += contentDelta;
    42→    42→
    43→    43→    const out: { reasoningDelta?: string; contentDelta?: string } = {};
    44→    44→    const OPEN = String.fromCharCode(60, 116, 104, 105, 110, 107, 62);   //  Dems
    45→    45→    const CLOSE = String.fromCharCode(60, 47, 116, 104, 105, 110, 107, 62); //  Democrats
    46→    46→
    47→    47→    // Loop: em um único feed podemos ter OPEN ... CLOSE ... OPEN ... CLOSE.
    48→    48→    // IMPORTANTE: depois de cada slice() no contentBuffer, tudo que sobra é
    49→    49→    // "não emitido por construção", então não rastreamos offsets — só
    50→    50→    // mantemos um lookahead de OPEN/CLOSE.length chars no fim (caso a tag
    51→    51→    // venha cortada entre dois chunks) e soltamos o resto no flush().
    52→    52→    while (true) {
    53→    53→      if (!this.inThinking) {
    54→    54→        const openIdx = this.contentBuffer.indexOf(OPEN);
    55→    55→        if (openIdx !== -1) {
    56→    56→          // Emite como content tudo que veio antes do OPEN.
    57→    57→          if (openIdx > 0) {
    58→    58→            out.contentDelta = (out.contentDelta || '') + this.contentBuffer.slice(0, openIdx);
    59→    59→          }
    60→    60→          this.contentBuffer = this.contentBuffer.slice(openIdx + OPEN.length);
    61→    61→          this.inThinking = true;
    62→    62→          this.thinkTagEmitted = true;
    63→    63→          continue; // volta pro while pra procurar CLOSE
    64→    64→        }
    65→    65→        // Não achou OPEN: emite tudo exceto os últimos OPEN.length chars
    66→    66→        // (lookahead de segurança, caso a tag venha cortada no próximo chunk).
    67→    67→        if (this.contentBuffer.length > OPEN.length) {
    68→    68→          const safeLen = this.contentBuffer.length - OPEN.length;
    69→    69→          out.contentDelta = (out.contentDelta || '') + this.contentBuffer.slice(0, safeLen);
    70→    70→          this.contentBuffer = this.contentBuffer.slice(safeLen);
    71→    71→        }
    72→    72→        break; // buffer pequeno e sem OPEN: espera próximo chunk
    73→    73→      }
    74→    74→
    75→    75→      // Estamos dentro de  Dems... Democrats: procura o fechamento.
    76→    76→      const closeIdx = this.contentBuffer.indexOf(CLOSE);
    77→    77→      if (closeIdx !== -1) {
    78→    78→        if (closeIdx > 0) {
    79→    79→          out.reasoningDelta = (out.reasoningDelta || '') + this.contentBuffer.slice(0, closeIdx);
    80→    80→        }
    81→    81→        this.contentBuffer = this.contentBuffer.slice(closeIdx + CLOSE.length);
    82→    82→        this.inThinking = false;
    83→    83→        continue; // volta pro while pra procurar próximo OPEN
    84→    84→      }
    85→    85→      // Não fechou ainda: emite tudo exceto os últimos CLOSE.length chars
    86→    86→      // (lookahead de segurança, caso a tag de fechamento venha cortada).
    87→    87→      if (this.contentBuffer.length > CLOSE.length) {
    88→    88→        const safeLen = this.contentBuffer.length - CLOSE.length;
    89→    89→        out.reasoningDelta = (out.reasoningDelta || '') + this.contentBuffer.slice(0, safeLen);
    90→    90→        this.contentBuffer = this.contentBuffer.slice(safeLen);
    91→    91→      }
    92→    92→      break; // buffer pequeno e sem CLOSE: espera próximo chunk
    93→    93→    }
    94→    94→    return out;
    95→    95→  }
    96→    96→
    97→    97→  /** Flush final: solta qualquer texto restante no buffer. */
    98→    98→  flush(): { reasoningDelta?: string; contentDelta?: string } {
    99→    99→    const out: { reasoningDelta?: string; contentDelta?: string } = {};
   100→   100→    if (this.inThinking) {
   101→   101→      // Nunca fechou  Dems: emite tudo como reasoning (caso patológico).
   102→   102→      if (this.contentBuffer) out.reasoningDelta = this.contentBuffer;
   103→   103→    } else {
   104→   104→      if (this.contentBuffer) out.contentDelta = this.contentBuffer;
   105→   105→    }
   106→   106→    this.contentBuffer = '';
   107→   107→    return out;
   108→   108→  }
   109→   109→}

   173→   173→      apiKey = process.env.NVIDIA_API_KEY;

   184→   184→

   204→   204→      return messages.map((m: any, index: number) => {
   205→   205→        if (m.role === 'tool') {
   206→   206→          if (!allowToolsInHistory) {
   207→   207→            return { role: 'user', content: `[Resultado da ferramenta ${m.tool_call_id || ''}]: ${m.content || ''}` };

   212→   212→          if (!allowToolsInHistory) {
   213→   213→            return { role: 'assistant', content: m.content || 'Executando ferramentas...' };

   216→   216→        }

   218→   218→          if (index === lastUserMsgIndex) {

   220→   220→              role: 'user',
   221→   221→              content: [
   222→   222→                { type: 'text', text: m.content || '' },

   224→   224→              ],
   225→   225→            };
   226→   226→          }
   227→   227→          return { role: 'user', content: `${m.content || ''}\n[Nota: ${m.images.length} imagem(ns) omitida(s)]` };
   228→   228→        }
   229→   229→        return { role: m.role, content: m.content || '' };
   230→   230→      });
   231→   231→    };
   232→   232→
   233→   233→    // Modelos NVIDIA que suportam tools: GLM-5.3 e GLM-5.3-Flash.

   236→   236→
   237→   237→    // ====== BUILDER DE PAYLOAD — NUVEM (NVIDIA) ======

   239→   239→      const hasToolsActive = includeTools && modelSupportsTools && tools && Array.isArray(tools) && tools.length > 0;
   240→   240→      const p: any = {
   241→   241→        model: selectedModel,
   242→   242→        messages: buildFormattedMessages(Boolean(hasToolsActive)),
   243→   243→        stream: true,
   244→   244→        max_tokens: safeTokenLimit ? 4096 : maxTokensFor(selectedEffort),
   245→   245→      };
   246→   246→      if (selectedModel === 'z-ai/glm-5.3') {
   247→   247→        p.stream_options = { include_usage: true };
   248→   248→      }
   249→   249→      if (includeReasoning && apiEffort && apiEffort !== 'none' && apiEffort !== 'default') {
   250→   250→        if (selectedModel === 'z-ai/glm-5.3') {
   251→   251→          p.reasoning_effort = apiEffort;
   252→   252→        }
   253→   253→      }
   254→   254→      if (hasToolsActive) {
   255→   255→        p.tools = tools;
   256→   256→        p.tool_choice = 'auto';
   257→   257→      }
   258→   258→      return p;
   259→   259→    };
   260→   260→
   261→   261→    // ====== BUILDER DE PAYLOAD — CLOUDFLARE WORKERS AI ======
   262→   262→    // O R1-Distill 32B da CF:
   263→   263→    //   - Não suporta reasoning_effort (o R1 sempre raciocina; o esforço é fixo).
   264→   264→    //   - Não suporta tools via endpoint OpenAI-compat.
   265→   265→    //   - Não suporta stream_options.include_usage.
   266→   266→    //   - max_tokens recomendado: 4096 (limite seguro para evitar cortes e custos altos).
   267→   267→    //   - Retorna o raciocínio embutido em delta.content entre <think>...</think>.
   268→   268→    //     O transformador CloudflareThinkSplitter (acima) cuida de separar isso.
   269→   269→    const createCloudflarePayload = () => {
   270→   270→      const p: any = {
   271→   271→        model: selectedModel,
   272→   272→        messages: buildFormattedMessages(false),
   273→   273→        stream: true,
   274→   274→        max_tokens: 4096,
   275→   275→        temperature: 0.6, // recomendado pela DeepSeek para R1-distill
   276→   276→      };
   277→   277→      return p;
   278→   278→    };
   279→   279→
   280→   280→    let payload = usingCloudflare
   281→   281→      ? createCloudflarePayload()
   282→   282→      : createNvidiaPayload(true, true, false);
   283→   283→
   284→   284→    console.log(
   285→   285→      `[chat] provider=${usingCloudflare ? 'cloudflare' : 'nvidia'} model=${selectedModel} effort=${selectedEffort}->${apiEffort} tools=${payload.tools?.length || 0}`
   286→   286→    );
   287→   287→
   288→   288→    const enc = new TextEncoder();
   289→   289→    const dec = new TextDecoder();
   290→   290→
   291→   291→    const stream = new ReadableStream({

   295→   295→

   303→   303→              controller.enqueue(enc.encode(': keepalive\n\n'));

   307→   307→          }

   309→   309→
   310→   310→        req.signal?.addEventListener('abort', () => {
   311→   311→          clearInterval(timer);
   312→   312→          try { controller.close(); } catch {}

   318→   318→            ? `https://api.cloudflare.com/client/v4/accounts/${cfAccountId}/ai/v1/chat/completions`

   332→   332→              method: 'POST',

   335→   335→              signal: fetchCtrl.signal,
   336→   336→            });
   337→   337→            clearTimeout(fetchTimer);
   338→   338→            return r;
   339→   339→          } catch (err) {
   340→   340→            clearTimeout(fetchTimer);
   341→   341→            throw err;
   342→   342→          }
   343→   343→        };
   344→   344→
   345→   345→        // ====== PIPE DE ERRO PADRÃO ======

   352→   352→                  choices: [{ index: 0, delta: { content: `\n\n⚠️ **Erro:** ${errMsg}` }, finish_reason: 'stop' }],
   353→   353→                })}\n\n`
   354→   354→              )

   356→   356→            controller.enqueue(enc.encode('data: [DONE]\n\n'));

   376→   376→              await res.body?.cancel().catch(() => {});
   377→   377→              payload = createNvidiaPayload(false, true, false);
   378→   378→              res = await fetchWithTimeout(payload);
   379→   379→            }
   380→   380→            // Fallback 2: 400/422 com tools -> tenta sem tools
   381→   381→            if (!res.ok && (res.status === 400 || res.status === 422) && payload.tools) {
   382→   382→              await res.body?.cancel().catch(() => {});
   383→   383→              payload = createNvidiaPayload(false, false, false);
   384→   384→              res = await fetchWithTimeout(payload);
   385→   385→            }
   386→   386→            // Fallback 3: 400/422 por excesso de max_tokens -> reduz para 4096

   388→   388→              await res.body?.cancel().catch(() => {});
   389→   389→              payload = createNvidiaPayload(false, false, true);
   390→   390→              res = await fetchWithTimeout(payload);
   391→   391→            }
   392→   392→          }
   393→   393→
   394→   394→          if (!res.ok || !res.body) {
   395→   395→            const errText = await res.text().catch(() => 'Erro desconhecido');

   397→   397→              ? `Erro da API Cloudflare (${res.status})`
   398→   398→              : `Erro da API NVIDIA (${res.status})`;

   400→   400→              const parsed = JSON.parse(errText);
   401→   401→              if (usingCloudflare && parsed?.errors?.length) {
   402→   402→                errMsg = parsed.errors.map((e: any) => e.message || JSON.stringify(e)).join('; ');
   403→   403→              } else if (parsed?.error?.message) {
   404→   404→                errMsg = parsed.error.message;
   405→   405→              } else if (parsed?.message) {
   406→   406→                errMsg = parsed.message;
   407→   407→              } else if (parsed?.detail) {
   408→   408→                errMsg = String(parsed.detail);
   409→   409→              }
   410→   410→            } catch {
   411→   411→              if (errText) errMsg = errText.slice(0, 300);
   412→   412→            }
   413→   413→            if (res.status === 401) {
   414→   414→              errMsg = usingCloudflare
   415→   415→                ? 'CLOUDFLARE_API_TOKEN inválido ou sem permissão de Workers AI.'
   416→   416→                : 'Chave NVIDIA_API_KEY inválida.';
   417→   417→            } else if (res.status === 429) {
   418→   418→              errMsg = 'Limite de requisições atingido (Rate Limit). Aguarde alguns segundos.';
   419→   419→            }
   420→   420→            sendErrorAndClose(errMsg);
   421→   421→            return;
   422→   422→          }
   423→   423→
   424→   424→          // ====== PIPE DOS DADOS REAIS ======
   425→   425→          // Se for Cloudflare, interceptamos o stream para separar <think> do content.
   426→   426→          const reader = res.body.getReader();
   427→   427→          let idleTimer: ReturnType<typeof setTimeout> | null = null;
   428→   428→          const resetIdleTimer = () => {
   429→   429→            if (idleTimer) clearTimeout(idleTimer);
   430→   430→            idleTimer = setTimeout(() => {
   431→   431→              console.log('[chat] IDLE TIMEOUT: provedor não mandou dados por 45s, abortando...');
   432→   432→              try { reader.cancel(); } catch {}
   433→   433→            }, 45000);
   434→   434→          };
   435→   435→          resetIdleTimer();
   436→   436→
   437→   437→          // Instanciado só quando necessário (CF). Para NVIDIA, é null e pulamos o rewrite.
   438→   438→          const splitter: CloudflareThinkSplitter | null = usingCloudflare ? new CloudflareThinkSplitter() : null;
   439→   439→          // Buffer de linha SSE bruta (para o caso de um chunk cortar no meio de uma linha)
   440→   440→          let sseBuffer = '';
   441→   441→
   442→   442→          try {
   443→   443→            while (true) {
   444→   444→              const { done, value } = await reader.read();
   445→   445→              if (done) break;
   446→   446→              resetIdleTimer();
   447→   447→              lastActivity = Date.now();
   448→   448→              const textChunk = dec.decode(value, { stream: true });
   449→   449→
   450→   450→              if (splitter) {
   451→   451→                const sp = splitter;
   452→   452→                // ====== MODO TRANSFORMADOR: parse SSE, separa <think>, reescreve ======
   453→   453→                sseBuffer += textChunk;
   454→   454→                isMidChunk = !sseBuffer.endsWith('\n');
   455→   455→                const lines = sseBuffer.split('\n');
   456→   456→                sseBuffer = lines.pop() || '';
   457→   457→
   458→   458→                for (const line of lines) {
   459→   459→                  const trimmed = line.trim();
   460→   460→                  if (!trimmed) continue;
   461→   461→                  if (trimmed.startsWith(':')) continue; // comentário keepalive
   462→   462→                  if (!trimmed.startsWith('data:')) continue;
   463→   463→
   464→   464→                  const dataStr = trimmed.replace(/^data:\s*/, '');
   465→   465→                  if (dataStr === '[DONE]') {
   466→   466→                    // Flush final antes de propagar o DONE
   467→   467→                    const flushed = sp.flush();
   468→   468→                    if (flushed.reasoningDelta) {
   469→   469→                      controller.enqueue(
   470→   470→                        enc.encode(
   471→   471→                          `data: ${JSON.stringify({
   472→   472→                            choices: [{ index: 0, delta: { reasoning_content: flushed.reasoningDelta } }],
   473→   473→                          })}\n\n`
   474→   474→                        )
   475→   475→                      );
   476→   476→                    }
   477→   477→                    if (flushed.contentDelta) {
   478→   478→                      controller.enqueue(
   479→   479→                        enc.encode(
   480→   480→                          `data: ${JSON.stringify({
   481→   481→                            choices: [{ index: 0, delta: { content: flushed.contentDelta } }],
   482→   482→                          })}\n\n`
   483→   483→                        )
   484→   484→                      );
   485→   485→                    }
   486→   486→                    controller.enqueue(enc.encode('data: [DONE]\n\n'));
   487→   487→                    continue;
   488→   488→                  }
   489→   489→
   490→   490→                  try {
   491→   491→                    const parsed = JSON.parse(dataStr);
   492→   492→                    const choice = parsed.choices?.[0];
   493→   493→                    const delta = choice?.delta || choice?.message;
   494→   494→
   495→   495→                    if (delta && typeof delta.content === 'string' && delta.content.length > 0) {
   496→   496→                      const split = sp.feed(delta.content);
   497→   497→                      // Constrói um novo chunk preservando tudo do original, só trocando o delta
   498→   498→                      const newDelta: any = { ...delta };
   499→   499→                      delete newDelta.content;
   500→   500→                      if (split.reasoningDelta) newDelta.reasoning_content = split.reasoningDelta;
   501→   501→                      if (split.contentDelta) newDelta.content = split.contentDelta;
   502→   502→
   503→   503→                      if (split.reasoningDelta || split.contentDelta) {
   504→   504→                        const newParsed = {
   505→   505→                          ...parsed,
   506→   506→                          choices: [{ ...choice, delta: newDelta }],
   507→   507→                        };
   508→   508→                        controller.enqueue(enc.encode(`data: ${JSON.stringify(newParsed)}\n\n`));
   509→   509→                      }
   510→   510→                      // Se nenhum dos dois veio (texto no buffer de lookahead), não emite nada
   511→   511→                      // ainda — vai ser emitido nos próximos chunks quando o lookahead fechar.
   512→   512→                    } else {
   513→   513→                      // Chunk sem content (ex.: só role, tool_calls, finish_reason) — repassa direto
   514→   514→                      controller.enqueue(enc.encode(`data: ${JSON.stringify(parsed)}\n\n`));
   515→   515→                    }
   516→   516→                  } catch {
   517→   517→                    // JSON malformado (raro): repassa a linha original
   518→   518→                    controller.enqueue(enc.encode(`${line}\n`));
   519→   519→                  }
   520→   520→                }
   521→   521→              } else {
   522→   522→                // ====== MODO PIPE DIRETO (NVIDIA) ======
   523→   523→                isMidChunk = !textChunk.endsWith('\n');
   524→   524→                controller.enqueue(value);
   525→   525→              }
   526→   526→            }
   527→   527→
   528→   528→            // Após o loop: se ainda há SSE buffer pendente no modo transformador, processa
   529→   529→            if (splitter && sseBuffer.trim()) {
   530→   530→              const spFinal = splitter;
   531→   531→              const trimmed = sseBuffer.trim();
   532→   532→              if (trimmed.startsWith('data:') && trimmed.replace(/^data:\s*/, '') === '[DONE]') {
   533→   533→                const flushed = spFinal.flush();
   534→   534→                if (flushed.reasoningDelta) {
   535→   535→                  controller.enqueue(
   536→   536→                    enc.encode(
   537→   537→                      `data: ${JSON.stringify({
   538→   538→                        choices: [{ index: 0, delta: { reasoning_content: flushed.reasoningDelta } }],
   539→   539→                      })}\n\n`
   540→   540→                    )
   541→   541→                  );
   542→   542→                }
   543→   543→                if (flushed.contentDelta) {
   544→   544→                  controller.enqueue(
   545→   545→                    enc.encode(
   546→   546→                      `data: ${JSON.stringify({
   547→   547→                        choices: [{ index: 0, delta: { content: flushed.contentDelta } }],
   548→   548→                      })}\n\n`
   549→   549→                    )
   550→   550→                  );
   551→   551→                }
   552→   552→                controller.enqueue(enc.encode('data: [DONE]\n\n'));
   553→   553→              } else if (trimmed.startsWith('data:')) {
   554→   554→                // Tenta processar como JSON normal (chunk final sem \n)
   555→   555→                try {
   556→   556→                  const dataStr = trimmed.replace(/^data:\s*/, '');
   557→   557→                  const parsed = JSON.parse(dataStr);
   558→   558→                  const choice = parsed.choices?.[0];
   559→   559→                  const delta = choice?.delta || choice?.message;
   560→   560→                  if (delta && typeof delta.content === 'string' && delta.content.length > 0) {
   561→   561→                    const split = spFinal.feed(delta.content);
   562→   562→                    const flushed = spFinal.flush();
   563→   563→                    const reasoningFull = (split.reasoningDelta || '') + (flushed.reasoningDelta || '');
   564→   564→                    const contentFull = (split.contentDelta || '') + (flushed.contentDelta || '');
   565→   565→                    const newDelta: any = { ...delta };
   566→   566→                    delete newDelta.content;
   567→   567→                    if (reasoningFull) newDelta.reasoning_content = reasoningFull;
   568→   568→                    if (contentFull) newDelta.content = contentFull;
   569→   569→                    if (reasoningFull || contentFull) {
   570→   570→                      controller.enqueue(
   571→   571→                        enc.encode(
   572→   572→                          `data: ${JSON.stringify({
   573→   573→                            ...parsed,
   574→   574→                            choices: [{ ...choice, delta: newDelta }],
   575→   575→                          })}\n\n`
   576→   576→                        )
   577→   577→                      );
   578→   578→                    }
   579→   579→                  } else {
   580→   580→                    controller.enqueue(enc.encode(`data: ${dataStr}\n\n`));
   581→   581→                  }
   582→   582→                } catch {
   583→   583→                  // ignora linha truncada irreconhecível
   584→   584→                }
   585→   585→              }
   586→   586→            }
   587→   587→          } finally {
   588→   588→            if (idleTimer) clearTimeout(idleTimer);
   589→   589→          }
   590→   590→          clearInterval(timer);
   591→   591→          controller.close();
   592→   592→        } catch (err: any) {
   593→   593→          clearInterval(timer);
   594→   594→          if (!req.signal?.aborted) {
   595→   595→            const errMsg = err?.message || 'Falha na conexão com a IA';
   596→   596→            sendErrorAndClose(errMsg);
   597→   597→          }
   598→   598→        }
   599→   599→      },
   600→   600→      async cancel() {},
   601→   601→    });
   602→   602→
   603→   603→    return new Response(stream, {
   604→   604→      status: 200,
   605→   605→      headers: {
   606→   606→        'Content-Type': 'text/event-stream; charset=utf-8',
   607→   607→        'Cache-Control': 'no-cache, no-transform',
   608→   608→        'Connection': 'keep-alive',
   609→   609→        'X-Accel-Buffering': 'no',
   610→   610→        'X-Content-Type-Options': 'nosniff',
   611→   611→      },
   612→   612→    });
   613→   613→  } catch (error: any) {
   614→   614→    console.error('Chat API error:', {
   615→   615→      message: error?.message,
   616→   616→      stack: error?.stack?.split('\n').slice(0, 5).join(' | '),
   617→   617→      name: error?.name,
   618→   618→    });
   619→   619→    const errMsg = error?.message || 'Erro ao processar mensagem.';
   620→   620→    return new Response(JSON.stringify({ error: errMsg, errorName: error?.name || 'Unknown' }), {
   621→   621→      status: 500,
   622→   622→      headers: { 'Content-Type': 'application/json' },
   623→   623→    });
   624→   624→  }
   625→   625→}
